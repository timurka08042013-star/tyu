local function isVisibleESP(targetChar, point)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LocalPlayer.Character, targetChar }
    params.IgnoreWater = true
    local origin = Camera.CFrame.Position
    local direction = point - origin
    local hit = workspace:Raycast(origin, direction, params)
    if not hit then return true end
    return (hit.Position - origin).Magnitude >= direction.Magnitude - 0.5
end

local function updateESP()
    for player, obj in pairs(espObjects) do
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local head = char and char:FindFirstChild("Head")
        local humanoid = char and char:FindFirstChildOfClass("Humanoid")

        if Features.ESPDeathCheck then
            if not (hrp and head and humanoid) or humanoid.Health <= 0 then
                obj.box.Visible = false
                continue
            end
        else
            if not (hrp and head) then
                obj.box.Visible = false
                continue
            end
        end

        if Features.ESPTeamCheck and player.Team == LocalPlayer.Team then
            obj.box.Visible = false
            continue
        end

        local myChar = LocalPlayer.Character
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if myHrp then
            local dist = (myHrp.Position - hrp.Position).Magnitude
            if dist > Features.ESPMaxDistance then
                obj.box.Visible = false
                continue
            end
        end

        local points = {
            head.Position + Vector3.new(0, 0.5, 0),
            hrp.Position + Vector3.new(0, 1, 0),
            hrp.Position - Vector3.new(0, 1.5, 0),
        }

        local visible = false
        for _, point in ipairs(points) do
            local pos, onScreen = Camera:WorldToViewportPoint(point)
            if onScreen and pos.Z > 0 then
                if isVisibleESP(char, point) then
                    visible = true
                    break
                end
            end
        end

        if not visible then
            obj.box.Visible = false
            continue
        end

        local topPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 1, 0))
        local bottomPos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))

        if not topPos or topPos.Z <= 0 or not bottomPos or bottomPos.Z <= 0 then
            obj.box.Visible = false
            continue
        end

        local height = math.abs(bottomPos.Y - topPos.Y) * 1.2
        local width = height * 0.7
        local x = topPos.X - width / 2
        local y = topPos.Y

        obj.box.Visible = true
        obj.box.Position = UDim2.new(0, x, 0, y)
        obj.box.Size = UDim2.new(0, width, 0, height)

        local color = getColor(player)
        obj.stroke.Color = color
        obj.box.BackgroundColor3 = color
        obj.nameLabel.Visible = Features.ESPName

        if Features.ESPHealth and humanoid then
            obj.healthBar.Visible = true
            local hp = humanoid.Health / humanoid.MaxHealth
            obj.healthBar.Size = UDim2.new(0, 3, hp, 0)
            obj.healthBar.Position = UDim2.new(0, -6, 1 - hp, 0)
            if hp > 0.6 then obj.healthBar.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
            elseif hp > 0.3 then obj.healthBar.BackgroundColor3 = Color3.fromRGB(255, 200, 0)
            else obj.healthBar.BackgroundColor3 = Color3.fromRGB(255, 0, 0) end
        else obj.healthBar.Visible = false end

        if Features.ESPDistance then
            obj.distLabel.Visible = true
            if myHrp then
                local dist = (myHrp.Position - hrp.Position).Magnitude
                obj.distLabel.Text = string.format("%d m", math.floor(dist))
            end
        else obj.distLabel.Visible = false end
    end
end

    if bypassEnabled then return end
    bypassEnabled = true
    pcall(function()
        local mt = getrawmetatable(game)
        local oldNamecall = mt.__namecall
        setreadonly(mt, false)

local function scalePart(part, multiplier, store)
    if not part or not part:IsA("BasePart") then return end
    if not store[part] then store[part] = part.Size end
    part.Size = store[part] * multiplier
    part.Transparency = 0.5
    part.CanCollide = false
    part.Massless = true
end

local function resetPart(part, store)
    if not part or not store[part] then return end
    part.Size = store[part]
    part.Transparency = 0
    part.CanCollide = true
    part.Massless = false
end

local function applyHitbox()
    for _, player in pairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        local char = player.Character
        if Features.HitboxDeathCheck then
            local humanoid = char and char:FindFirstChildOfClass("Humanoid")
            if not char or not humanoid or humanoid.Health <= 0 then
                if originalSizes[player] then
                    local head = char and char:FindFirstChild("Head")
                    local torso = char and (char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"))
                    local hrp = char and char:FindFirstChild("HumanoidRootPart")
                    local store = originalSizes[player]
                    if head then resetPart(head, store) end
                    if torso then resetPart(torso, store) end
                    if hrp then resetPart(hrp, store) end
                end
                continue
            end
        else
            if not char then continue end
        end
        if Features.HitboxTeamCheck and player.Team == LocalPlayer.Team then
            if originalSizes[player] then
                local head = char:FindFirstChild("Head")
                local torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local store = originalSizes[player]
                if head then resetPart(head, store) end
                if torso then resetPart(torso, store) end
                if hrp then resetPart(hrp, store) end
            end
            continue
        end
        local head = char:FindFirstChild("Head")
        local torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not originalSizes[player] then originalSizes[player] = {} end
        local store = originalSizes[player]
        if Features.Hitbox then
            if head then scalePart(head, Features.HeadSize, store) end
            if torso then scalePart(torso, Features.TorsoSize, store) end
            if hrp then scalePart(hrp, Features.HitboxSize, store) end
        else
            if head then resetPart(head, store) end
            if torso then resetPart(torso, store) end
            if hrp then resetPart(hrp, store) end
        end
    end
end

            local rPart = char:FindFirstChild(Radon.RageHitPart) or targetPart
            local alive = true
            if Radon.RageVisibleCheck and not onScreen then alive = false end
            if alive and Radon.RageWallCheck and not isVisibleRadon(rPart) then alive = false end
            if alive then
                local rd = (lHeadPos - rPart.Position).Magnitude
                if rd < rDist then rDist = rd rClose = rPart end
            end
        end

        if Radon.SilentAim and onScreen then
            if not Radon.SilentTeamCheck or not isTeam then
                local sd = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                if sd <= (Radon.SilentUseFov and Radon.SilentFov or 999999) then
                    if sd < sDist then

                            if not isVisibleRadon(targetPart) then continue end
                        end
                        sDist = sd
                        sClose = targetPart
                    end
                end
            end
        end
    end

    Radon.SilentTarget = sClose
    Radon.RageTarget = rClose
end

local ChamsSettings = {
    Enabled = false,
    VisibleColor = Color3.fromRGB(0, 255, 0),
    InvisibleColor = Color3.fromRGB(255, 0, 0),
    VisibleTransparency = 0,
    InvisibleTransparency = 0.7,
    TeamCheck = true,
    MaxDistance = 500,
}

local ChamsFolder = Instance.new("Folder")
ChamsFolder.Name = "BloxStrikeChams"
ChamsFolder.Parent = game.CoreGui

local chamsObjects = {}

local function getChamsColor(player)
    if ChamsSettings.TeamCheck and player.Team == LocalPlayer.Team then
        return TEAM_COLOR
    end
    return ENEMY_COLOR
end

local function isVisibleChams(targetChar)
    if not targetChar or not targetChar:FindFirstChild("HumanoidRootPart") then return false end
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("Head") then return false end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {myChar}
    params.IgnoreWater = true
    local origin = Camera.CFrame.Position
    local target = targetChar.HumanoidRootPart.Position
    local hit = workspace:Raycast(origin, target - origin, params)
    if not hit then return true end
    return hit.Instance:IsDescendantOf(targetChar)
end

local function createChams(player)
    if player == LocalPlayer then return end
    if chamsObjects[player] then return end
    local char = player.Character
    if not char then return end
    local chams = { boxes = {} }
    for _, part in pairs(char:GetChildren()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            local box = Instance.new("BoxHandleAdornment")
            box.Name = "Cham_" .. part.Name
            box.Adornee = part
            box.AlwaysOnTop = true
            box.ZIndex = 5
            box.Transparency = 0
            box.Color3 = getChamsColor(player)
            box.Size = part.Size
            box.Parent = ChamsFolder
            table.insert(chams.boxes, { box = box, part = part, player = player })
        end
    end
    chamsObjects[player] = chams
end

local function removeChams(player)
    if chamsObjects[player] then
        for _, data in pairs(chamsObjects[player].boxes) do
            if data.box then data.box:Destroy() end
        end
        chamsObjects[player] = nil
    end
end

local function updateChams()
    for player, chams in pairs(chamsObjects) do
        local char = player.Character
        if not char then continue end
        local myChar = LocalPlayer.Character
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if myHrp and hrp then
            local dist = (myHrp.Position - hrp.Position).Magnitude
            if dist > ChamsSettings.MaxDistance then
                for _, data in pairs(chams.boxes) do data.box.Transparency = 1 end
                continue
            end
        end
        if ChamsSettings.TeamCheck and player.Team == LocalPlayer.Team then
            for _, data in pairs(chams.boxes) do data.box.Transparency = 1 end
            continue
        end
        local visible = isVisibleChams(char)
        for _, data in pairs(chams.boxes) do
            if data.box and data.part and data.part.Parent then
                data.box.Size = data.part.Size
                if visible then
                    data.box.Color3 = ChamsSettings.VisibleColor
                    data.box.Transparency = ChamsSettings.VisibleTransparency
                else
                    data.box.Color3 = ChamsSettings.InvisibleColor
                    data.box.Transparency = ChamsSettings.InvisibleTransparency
                end
            end
        end
    end
end

local TracersSettings = {
    Enabled = false, Origin = "Bottom", Thickness = 1, Length = 500,
    Color = Color3.fromRGB(255, 255, 255), TeamCheck = true, MaxDistance = 500,
}

local TracersFolder = Instance.new("Folder")
TracersFolder.Name = "BloxStrikeTracers"
TracersFolder.Parent = Camera

local tracerObjects = {}

local function getTracerOrigin()
    local vp = Camera.ViewportSize
    if TracersSettings.Origin == "Bottom" then return Vector2.new(vp.X / 2, vp.Y)
    elseif TracersSettings.Origin == "Center" then return Vector2.new(vp.X / 2, vp.Y / 2)
    elseif TracersSettings.Origin == "Mouse" then
        local m = UserInputService:GetMouseLocation()
        return Vector2.new(m.X, m.Y)
    end
    return nil
end

local function createTracer(player)
    if player == LocalPlayer then return end
    if tracerObjects[player] then return end
    local part = Instance.new("Part")
    part.Name = player.Name .. "_Tracer"
    part.Material = Enum.Material.Neon
    part.Transparency = 1
    part.Anchored = true
    part.CanCollide = false
    part.Locked = true
    part.Parent = TracersFolder
    local box = Instance.new("BoxHandleAdornment")
    box.Name = "TracerBox"
    box.Adornee = part
    box.AlwaysOnTop = true
    box.ZIndex = 5
    box.Transparency = 0
    box.Color3 = TracersSettings.Color
    box.Size = Vector3.new(0.05, 0.05, 1)
    box.Parent = part
    tracerObjects[player] = { part = part, box = box }
end

local function removeTracer(player)
    if tracerObjects[player] then
        if tracerObjects[player].part then tracerObjects[player].part:Destroy() end
        tracerObjects[player] = nil
    end
end

local function updateTracers()
    for player, data in pairs(tracerObjects) do
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then
            data.box.Transparency = 1
            data.part.Transparency = 1
            continue
        end
        if TracersSettings.TeamCheck and player.Team == LocalPlayer.Team then
            data.box.Transparency = 1
            continue
        end
        local myChar = LocalPlayer.Character
        local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
        if myHrp then
            local dist = (myHrp.Position - hrp.Position).Magnitude
            if dist > TracersSettings.MaxDistance then
                data.box.Transparency = 1
                continue
            end
        end
        local origin = getTracerOrigin()
        local fromPos
        if origin then
            local ray = Camera:ScreenPointToRay(origin.X, origin.Y)
            fromPos = ray.Origin
        else
            fromPos = myHrp and myHrp.Position or Camera.CFrame.Position
        end
        local toPos = hrp.Position
        local dir = (toPos - fromPos)
        local length = math.min(dir.Magnitude, TracersSettings.Length)
        dir = dir.Unit * length
        data.part.Size = Vector3.new(0.05, 0.05, length)
        data.part.CFrame = CFrame.new(fromPos + dir / 2, toPos)
        data.box.Size = Vector3.new(TracersSettings.Thickness * 0.05, TracersSettings.Thickness * 0.05, length)
        data.box.Color3 = TracersSettings.Color
        data.box.Transparency = 0
    end
end

local function applyConfig(config)
    if not config then return end
    if config.Features then
        for k, v in pairs(config.Features) do Features[k] = v end
    end
    if config.Radon and Radon then
        for k, v in pairs(config.Radon) do Radon[k] = v end
    end
    if config.World and WorldSettings then
        for k, v in pairs(config.World) do WorldSettings[k] = v end
        if applyFullbright then applyFullbright() end
        if applyNoFog then applyNoFog() end
        if applyTime then applyTime() end
        if applySky then applySky() end
    end
    if config.Crosshair and CrosshairSettings then
        for k, v in pairs(config.Crosshair) do CrosshairSettings[k] = v end
        if updateCrosshair then updateCrosshair() end
    end
    if config.Chams and ChamsSettings then
        for k, v in pairs(config.Chams) do ChamsSettings[k] = v end
    end
    if config.Tracers and TracersSettings then
        for k, v in pairs(config.Tracers) do TracersSettings[k] = v end
    end
    if config.Movement and MovementSettings then
        for k, v in pairs(config.Movement) do MovementSettings[k] = v end
    end
    if config.Fun and FunSettings then
        for k, v in pairs(config.Fun) do FunSettings[k] = v end
    end
end

local function saveConfig(name)
    if not hasFileSystem then
        warn("[Config] Инжектор не поддерживает writefile")
        return false
    end
    local data = collectConfig()
    local ok, encoded = pcall(function() return HttpService:JSONEncode(data) end)
    if not ok then return false end
    return pcall(function() writefile(getConfigPath(name), encoded) end)
end

local function loadConfig(name)
    if not hasFileSystem then
        warn("[Config] Инжектор не поддерживает readfile")
        return false
    end
    local path = getConfigPath(name)
    if not isfile(path) then
        warn("[Config] Файл не найден:", path)
        return false
    end
    local ok, data = pcall(function() return readfile(path) end)
    if not ok then return false end
    local ok2, decoded = pcall(function() return HttpService:JSONDecode(data) end)
    if not ok2 then return false end
    applyConfig(decoded)
    return true
end

local function listConfigs()
    if not hasFileSystem then return {} end
    local list = {}
    local ok, files = pcall(function() return listfiles(ConfigSystem.FolderName) end)
    if not ok then return {} end
    for _, path in pairs(files) do
        local name = path:match("([^/\\]+)%.json$")
        if name then table.insert(list, name) end
    end
    return list
end

local function deleteConfig(name)
    if not hasFileSystem then return false end
    local path = getConfigPath(name)
    if not isfile(path) then return false end
    return pcall(function() delfile(path) end)
end

-- ============ KEYBIND SYSTEM ============
local KeybindSystem = {
    Binds = {},
}

local function RegisterKeybind(name, key, mode, callback)
    KeybindSystem.Binds[name] = {
        key = key, mode = mode or "Toggle", state = false, callback = callback,
    }
end

local function UpdateKeybind(name, key, mode)
    if KeybindSystem.Binds[name] then
        KeybindSystem.Binds[name].key = key
        if mode then KeybindSystem.Binds[name].mode = mode end
    end
end

local function matchesBind(input, key)
    if not key or not input then return false end
    if typeof(key) == "EnumItem" then
        if key.EnumType == Enum.KeyCode then return input.KeyCode == key
        elseif key.EnumType == Enum.UserInputType then return input.UserInputType == key end
    end
    return false
end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    for name, bind in pairs(KeybindSystem.Binds) do
        if matchesBind(input, bind.key) then
            if bind.mode == "Toggle" then
                bind.state = not bind.state
            else
                bind.state = true
            end
            if bind.callback then bind.callback(bind.state) end
        end
    end
end)

UserInputService.InputEnded:Connect(function(input, gp)
    if gp then return end
    for name, bind in pairs(KeybindSystem.Binds) do
        if matchesBind(input, bind.key) and bind.mode == "Hold" then
            bind.state = false
            if bind.callback then bind.callback(false) end
        end
    end
end)

    local function createSlider(parent, text, min, max, default, callback)
        local row = Instance.new("Frame")
        row.Size = UDim2.new(1, 0, 0, 42)
        row.BackgroundTransparency = 1
        row.Parent = parent
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, -50, 0, 16)
        label.BackgroundTransparency = 1
        label.Text = text
        label.TextColor3 = Colors.Text
        label.Font = Enum.Font.GothamMedium
        label.TextSize = 12
        label.TextXAlignment = Enum.TextXAlignment.Left
        label.Parent = row
        local valueLabel = Instance.new("TextLabel")
        valueLabel.Size = UDim2.new(0, 50, 0, 16)
        valueLabel.Position = UDim2.new(1, -50, 0, 0)
        valueLabel.BackgroundTransparency = 1
        valueLabel.Text = tostring(default)
        valueLabel.TextColor3 = isPremium and Colors.Gold or Colors.Accent
        valueLabel.Font = Enum.Font.GothamSemibold
        valueLabel.TextSize = 12
        valueLabel.TextXAlignment = Enum.TextXAlignment.Right
        valueLabel.Parent = row
        local bar = Instance.new("Frame")
        bar.Size = UDim2.new(1, 0, 0, 6)
        bar.Position = UDim2.new(0, 0, 0, 28)
        bar.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
        bar.BorderSizePixel = 0
        bar.Parent = row
        Instance.new("UICorner", bar).CornerRadius = UDim.new(1, 0)
        local fill = Instance.new("Frame")
        fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
        fill.BackgroundColor3 = isPremium and Colors.Gold or Colors.Accent
        fill.BorderSizePixel = 0
        fill.Parent = bar
        Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, 0, 1, 0)
        btn.BackgroundTransparency = 1
        btn.Text = ""
        btn.Parent = row
        local dragging = false
        local function update(input)
            local rel = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
            local val = math.floor(min + (max - min) * rel)
            fill.Size = UDim2.new(rel, 0, 1, 0)
            valueLabel.Text = tostring(val)
            if callback then callback(val) end
        end
        btn.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = true update(input) end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then update(input) end
        end)
        UserInputService.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
        end)
        return row
    end

    local function makeSubTabs(parentPage, tabNames)
        local bar = Instance.new("Frame")
        bar.Size = UDim2.new(1, 0, 0, 34)
        bar.BackgroundColor3 = Colors.Card
        bar.BorderSizePixel = 0
        bar.LayoutOrder = -1
        bar.Parent = parentPage
        Instance.new("UICorner", bar).CornerRadius = UDim.new(0, 8)
        local pad = Instance.new("UIPadding")
        pad.PaddingLeft = UDim.new(0, 6)
        pad.PaddingRight = UDim.new(0, 6)
        pad.Parent = bar
        local layout = Instance.new("UIListLayout")
        layout.FillDirection = Enum.FillDirection.Horizontal
        layout.Padding = UDim.new(0, 4)
        layout.SortOrder = Enum.SortOrder.LayoutOrder
        layout.VerticalAlignment = Enum.VerticalAlignment.Center
        layout.Parent = bar
        local pagesHolder = Instance.new("Frame")
        pagesHolder.Size = UDim2.new(1, 0, 0, 0)
        pagesHolder.AutomaticSize = Enum.AutomaticSize.Y
        pagesHolder.BackgroundTransparency = 1
        pagesHolder.LayoutOrder = 1
        pagesHolder.Parent = parentPage
        local pagesLayout = Instance.new("UIListLayout")
        pagesLayout.Padding = UDim.new(0, 8)
        pagesLayout.SortOrder = Enum.SortOrder.LayoutOrder
        pagesLayout.Parent = pagesHolder
        local store = {}
        local current = nil
        local function select(name)
            for n, t in pairs(store) do
                local active = (n == name)
                t.page.Visible = active
                TweenService:Create(t.button, TweenInfo.new(0.12), {
                    BackgroundColor3 = active and (isPremium and Colors.Gold or Colors.Accent) or Color3.fromRGB(30, 30, 35),
                    TextColor3 = active and Color3.fromRGB(255, 255, 255) or Colors.SubText,
                }):Play()
            end
            current = name
        end
        for i, name in ipairs(tabNames) do
            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(0, 0, 0, 24)
            btn.AutomaticSize = Enum.AutomaticSize.X
            btn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
            btn.Text = name
            btn.TextColor3 = Colors.SubText
            btn.Font = Enum.Font.GothamMedium
            btn.TextSize = 11
            btn.BorderSizePixel = 0
            btn.AutoButtonColor = false
            btn.LayoutOrder = i
            btn.Parent = bar
            Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)
            local p = Instance.new("UIPadding", btn)
            p.PaddingLeft = UDim.new(0, 12)
            p.PaddingRight = UDim.new(0, 12)
            local page = Instance.new("Frame")
            page.Size = UDim2.new(1, 0, 0, 0)
            page.BackgroundTransparency = 1
            page.AutomaticSize = Enum.AutomaticSize.Y
            page.Visible = false
            page.LayoutOrder = i
            page.Parent = pagesHolder
            local pl = Instance.new("UIListLayout")
            pl.Padding = UDim.new(0, 8)
            pl.SortOrder = Enum.SortOrder.LayoutOrder
            pl.Parent = page
            store[name] = { button = btn, page = page }
            btn.MouseButton1Click:Connect(function() select(name) end)
            btn.MouseEnter:Connect(function()
                if current ~= name then
                    TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundColor3 = Color3.fromRGB(40, 40, 48) }):Play()
                end
            end)
            btn.MouseLeave:Connect(function()
                if current ~= name then
                    TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundColor3 = Color3.fromRGB(30, 30, 35) }):Play()
                end
            end)
        end
        select(tabNames[1])
        return store, select
    end
